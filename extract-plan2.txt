function planGuestAnswers(){
  try {
    const raw = localStorage.getItem(PLAN_GUEST_KEY);
    if (!raw) return null;
    const d = JSON.parse(raw);
    return (d && (d.problems || []).length) ? d : null;
  } catch (e) { return null; }
}
function planGuestSave(answers){
  try { localStorage.setItem(PLAN_GUEST_KEY, JSON.stringify(answers || {})); } catch (e) {}
}

async function loadPlanAnswers(){
  /* ⚠️⚠️ ОТМЕТКУ СТАВИМ ВО ВСЕХ ИСХОДАХ, ВКЛЮЧАЯ НЕУДАЧНЫЕ. Она значит
     «СПРАШИВАТЬ БОЛЬШЕ НЕ НАДО», а не «ответ получен». Раньше выход по
     «нет telegram_id» (гость в вебе), по коду ответа и по исключению
     оставлял её ложной — а `renderPlanScreen` в этом случае звал загрузку
     и перерисовывался снова: приложение вставало намертво (поймал Denis
     20.08 в демо, кнопка «Составить план»). Классика: флаг «данные не
     приехали» перепутан с флагом «данных нет» */
  try {
    /* ⚠️ У ГОСТЯ КАРТА ЖИВЁТ НА УСТРОЙСТВЕ. Ему негде храниться на сервере —
       телеграма нет, а карту он собирает сам и она должна пережить закрытие
       вкладки. При опознании (вход, подписка на бота) её надо будет перенести
       на сервер — точка переноса `planGuestAnswers()` */
    if (!telegramId){ planAnswers = planGuestAnswers(); return; }
    const r = await fetch(`${API_URL}/plan-answers/${telegramId}`);
    if (!r.ok) return;
    const d = await r.json();
    planAnswers = d.exists ? { problems: d.problems || [], level: d.level || '',
                               depth: d.depth || '', note: d.note || '' } : null;
  } catch(e){
    console.warn('[личный план] ответы не приехали:', e && e.message || e);
  } finally {
    planAnswersLoaded = true;
  }
}

function planLecturesFor(key, answers, used){
  const dict = planProblems.find(function(x){ return x.key === key; });
  const keys = [key];
  if (dict && dict.parent_key) keys.push(dict.parent_key);

  const out = [];
  keys.forEach(function(k){
    if (out.length >= PLAN_PER_PROBLEM) return;
    const rows = [];
    allLectures.forEach(function(lec){
      if (used[lec.slug]) return;                    // одна лекция не попадает в план дважды
      const entry = planEntryFor(lec, k);
      if (!entry) return;
      rows.push({ lec: lec, entry: entry, score: planScore(lec, entry, answers),
                  viaParent: k !== key });
    });
    rows.sort(function(a, b){ return b.score - a.score; });
    rows.slice(0, PLAN_PER_PROBLEM - out.length).forEach(function(r){
      used[r.lec.slug] = true;
      out.push(r);
    });
  });
  return out;
}

/* Catmull-Rom проходит РОВНО через снятые точки, в отличие от Безье,
   где они были бы лишь управляющими */
function planSpline(p, t){
  const n = p.length, s = t * (n - 1);
  const i = Math.min(Math.floor(s), n - 2), u = s - i;
  const a = p[Math.max(i-1,0)], b = p[i], c = p[i+1], d = p[Math.min(i+2,n-1)];
  const f = (A,B,C,D) => 0.5*((2*B) + (-A+C)*u + (2*A-5*B+4*C-D)*u*u + (-A+3*B-3*C+D)*u*u*u);
  return { x: f(a[0],b[0],c[0],d[0]), y: f(a[1],b[1],c[1],d[1]) };
}

/* Раскладка РАВНОМЕРНО ПО ДЛИНЕ ДУГИ, а не по параметру t: иначе на изгибах
   остановки сбивались бы в кучу, а на прямых разъезжались */
function planAlong(p, count){
  const S = 400, samp = [], len = [0];
  let tot = 0;
  for (let i = 0; i <= S; i++) samp.push(planSpline(p, i / S));
  for (let j = 1; j <= S; j++){
    const dx = samp[j].x - samp[j-1].x, dy = samp[j].y - samp[j-1].y;
    tot += Math.sqrt(dx*dx + dy*dy); len.push(tot);
  }
  const out = []; let k = 0;
  for (let m = 0; m < count; m++){
    const want = count === 1 ? 0 : tot * m / (count - 1);
    while (k < S && len[k+1] < want) k++;
    out.push(samp[k]);
  }
  return out;
}

/* Высота тропы ленты в точке t (0..100 по ширине плитки) */
function planRailY(t){
  const p = PLAN_RAIL_PATH[planTheme()];
  for (let i = 1; i < p.length; i++){
    if (p[i][0] >= t){
      const a = p[i-1], b = p[i], u = (t - a[0]) / ((b[0] - a[0]) || 1);
      return a[1] + (b[1] - a[1]) * u;
    }
  }
  return p[p.length-1][1];
}

/* Прогрессивная прозрачность будущих. Считается ОТ СЛЕДУЮЩЕЙ, а не от первой
   лекции — иначе по мере прохождения карта выцветала бы целиком. Диапазон
   растягивается под число остановок: следующая всегда 1.0, самая дальняя
   всегда PLAN_ALPHA_MIN, и эффект одинаково читается при 4 и при 10 */
function planFutureAlpha(k, last){
  if (k <= 0 || last <= 0) return 1;
  return +(1 - (1 - PLAN_ALPHA_MIN) * (k / last)).toFixed(3);
}

/* ПЕРСПЕКТИВА: чем выше остановка на карте, тем она мельче */
function planPersp(y, topY){
  let t = (100 - y) / Math.max(1, 100 - topY);
  t = Math.max(0, Math.min(1, t));
  return +(1 - (1 - PLAN_PERSP_MIN) * t).toFixed(3);
}

   растягивается под число остановок: следующая всегда 1.0, самая дальняя
   всегда PLAN_ALPHA_MIN, и эффект одинаково читается при 4 и при 10 */
function planFutureAlpha(k, last){
  if (k <= 0 || last <= 0) return 1;
  return +(1 - (1 - PLAN_ALPHA_MIN) * (k / last)).toFixed(3);
}

/* ПЕРСПЕКТИВА: чем выше остановка на карте, тем она мельче */
function planPersp(y, topY){
  let t = (100 - y) / Math.max(1, 100 - topY);
  t = Math.max(0, Math.min(1, t));
  return +(1 - (1 - PLAN_PERSP_MIN) * t).toFixed(3);
}

/* Сторона и ширина подписи — по СВОБОДНОМУ МЕСТУ до края арта, а не «через
   раз»: у края пейзажа подпись иначе вылезает наружу */
function planCapBox(x){
  const half = 10;
  const right = 100 - (x + half) - 2, left = (x - half) - 2;
  const side = right >= left ? 'side-r' : 'side-l';
  return { side: side, w: Math.min(46, Math.max(22, side === 'side-r' ? right : left)) };
}

function planImgVariant(seed){
  let h = 0;
  const str = String(seed || '');
  for (let i = 0; i < str.length; i++) h = (h * 31 + str.charCodeAt(i)) >>> 0;
  return (h % PLAN_IMG_VARIANTS) + 1;
}

function planRailY(t){
  const p = PLAN_RAIL_PATH[planTheme()];
  for (let i = 1; i < p.length; i++){
    if (p[i][0] >= t){
      const a = p[i-1], b = p[i], u = (t - a[0]) / ((b[0] - a[0]) || 1);
      return a[1] + (b[1] - a[1]) * u;
    }
  }
  return p[p.length-1][1];
}

/* Попап остановки. Сознательно бедный: обложка, название, кнопки. Описание
   и теги остаются в библиотеке, где человек выбирает сам */
function openPlanStop(i){
  const s = planStops[i];
  if (!s) return;
  const el = document.getElementById('plan-stop-body');
  const stepEl = document.getElementById('plan-stop-step');
  if (!el) return;
  if (stepEl) stepEl.textContent = 'Шаг ' + (i + 1) + ' из ' + planStops.length;

  const color = planStopColor(s, i);
  const notes = [];
  if (s.entry && s.entry.timecode) notes.push('про это на ' + escapeHtml(s.entry.timecode));
  if (s.entry && s.entry.weight === 'touch') notes.push('затронуто вскользь');
  if (s.viaParent) notes.push('разбор темы шире');

  const l = s.lec;
  const hasFree = planStopHasFree(s);
  /* ⚠️ ГОСТЮ ГОВОРИМ ПРАВДУ, А НЕ ПРЯЧЕМ ЗАНЯТИЕ. [stated] Denis 20.08:
     «не стесняться и показывать, что лекция на эту тему есть, но бесплатной
     части у неё нет. Хочешь — покупай». Закрытая остановка — доказательство,
     что клуб закрывает его боль, а не насмешка. Замок в прежнем смысле
     остаётся только участнику во время знакомства */
  const locked = isMember ? tourActive : !hasFree;
  const lockIcon = locked ? '🔒 ' : '';
  const hasNext = i + 1 < planStops.length;

  let acts;
  if (!isMember && !hasFree){
    // Открытой части нет — не рисуем кнопки-обманки, а прямо зовём в клуб
    acts = '<div class="pstop-note" style="margin-top:10px">'
        + escapeHtml(t('plan.guest_shut') || '') + '</div>'
      + '<button class="lec-link-btn access-btn" style="margin-top:10px" '
        + 'onclick="planStopBuy(' + i + ')">' + escapeHtml(t('plan.guest_buy') || 'Открыть доступ') + '</button>';
  } else {
    const label = (!isMember && hasFree) ? (t('plan.guest_free') || 'Смотреть открытую часть') : '';
    acts = '<div class="lec-links">'
      + (l.link ? '<button class="lec-link-btn' + (locked ? ' locked' : '') + '" onclick="planStopWatch(' + i + ',\'tg\')">' + lockIcon + (label || 'Смотреть в ТГ') + '</button>' : '')
      + ((l.link_vk && isMember) ? '<button class="lec-link-btn' + (locked ? ' locked' : '') + '" onclick="planStopWatch(' + i + ',\'vk\')">' + lockIcon + 'Смотреть в VK</button>' : '')
      + '</div>';
  }

  el.innerHTML =
      '<img class="pstop-img" src="images/lectures/' + l.slug + '.jpg" alt="" '
    + 'onerror="this.outerHTML=\'<div class=&quot;pstop-img-fallback&quot; style=&quot;--pc:' + color + '&quot;></div>\'">'
    + '<div class="pstop-title">' + escapeHtml(l.title || '') + '</div>'
    + '<div class="pstop-note">' + escapeHtml(s.group || '')
      + (notes.length ? ' · ' + notes.join(' · ') : '') + '</div>'
    + acts
    + (hasNext ? '<button class="lec-link-btn access-btn" style="margin-top:12px" onclick="planStopNext(' + i + ')">Открыть следующую</button>' : '')
    + '<button class="pstop-skip" onclick="planStopSkip(' + i + ')">Это не хочу смотреть</button>';

  document.getElementById('plan-stop-overlay').classList.add('open');
}

  const el = document.getElementById('plan-inner');
  if (!el) return;
  // Ответы ещё не спрашивали — сначала спросим, потом решим, что показывать
  if (!planAnswersLoaded){
    /* ⚠️ ВТОРОЙ ЗАМОК ОТ ТОЙ ЖЕ ПЕТЛИ: даже если отметка почему-то снова не
       встанет, повторно грузить и перерисовывать себя мы не будем */
    el.innerHTML = '<div class="loading">Загрузка...</div>';
    if (!window._planAnswersAsking){
      window._planAnswersAsking = true;
      loadPlanAnswers().then(function(){
        window._planAnswersAsking = false;
        planAnswersLoaded = true;
        renderPlanScreen();
      });
    }
    return;
  }
  if (!planHasAnything()){
    /* ⚠️ ВО ВРЕМЯ ЗНАКОМСТВА ЭКРАН НЕ ЗАПУСКАЕТ АНКЕТУ САМ: поле выбора тем
       полноэкранное, оно съедало бы шаг целиком ещё до того, как человек
       прочитал, о чём он ([stated] Denis 19.08). Поле открывает кнопка шага */
    if (onbActive && !onbTrying){
      el.innerHTML = '<div class="empty">' + escapeHtml(t('onb.plan_empty')
        || 'Пути пока нет — соберём его прямо сейчас.') + '</div>';
      return;
    }
    // Справочник ещё едет — иначе мелькнёт «пока нечего выбрать»
    if (!planProblems.length){
      el.innerHTML = '<div class="loading">Загрузка...</div>';
      loadPlanProblems().then(renderPlanQuestion);
      return;
    }
    renderPlanQuestion();
    return;
  }
  el.innerHTML = planMapHtml(planAnswersSafe())
    + '<div class="payment-help-link" style="margin-top:16px" onclick="startPlanQuiz()">'
    + (planHasThemes() ? 'Пересобрать план →' : 'Добавить свои темы →') + '</div>';
}

     ровно столько, сколько считает его телефон. [stated] Denis: «важно, чтобы

     Порядок: показать поле с надписью → отдать браузеру кадр → считать. */
  const wait = document.getElementById('pf-wait');
  if (wait) wait.style.display = 'flex';
  f.classList.add('show');
  planFieldChrome(true);
  requestAnimationFrame(function(){ f.classList.add('in'); });

  /* Два кадра, а не один: первый ставит стили, второй их рисует */
  requestAnimationFrame(function(){
    requestAnimationFrame(function(){
      try { pfRender(); } finally {
        if (wait) wait.style.display = 'none';
      }
      /* Замеряем плавность именно здесь: самый тяжёлый экран клуба */
      if (typeof perfWatch === 'function') perfWatch(2000);
    });
  });

  const dock = f.querySelector('.pf-dock');
  if (dock){
    // Перезапуск анимации: класс мог остаться с прошлого открытия, и тогда
    // браузер посчитал бы её уже отыгранной
    dock.classList.remove('born');
    void dock.offsetWidth;
    dock.classList.add('born');
  }
}

async function finishPlanQuiz(){
  onbDone('plan');
  const noteEl = document.getElementById('plan-note');
  planDraft.note = noteEl ? noteEl.value.trim() : '';
  planAnswers = {
    problems: planDraft.problems.slice(), level: planDraft.level,
    depth: planDraft.depth, note: planDraft.note,
  };
  renderPlanScreen();
  window.scrollTo(0, 0);
  renderPersonalPlanBody();
  // Шаг уровня «Построить личный путь» до сих пор закрывался заглушкой —
  // теперь его закрывает настоящее прохождение анкеты
  if (!builtPath) placeholderBuildPathStep();
  // Сохраняем после показа: план человек уже видит, а сеть подождёт
  if (!telegramId) planGuestSave(planAnswers);
  if (telegramId){
    fetch(`${API_URL}/plan-answers`, {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        telegram_id: telegramId, problems: planAnswers.problems,
        level: planAnswers.level, depth: planAnswers.depth, note: planAnswers.note,
      })
    }).catch(function(){});
  }
}
